{%- comment -%} Sets `safe_href` from include.url. Allows https/http links, mailto:, #anchors and site-relative paths; anything else (e.g. a javascript: link pasted by mistake) becomes "#". Sets `is_external` for links that leave the site (they open in a new tab). {%- endcomment -%} {%- assign safe_href = include.url | default: "" | strip -%} {%- assign low = safe_href | downcase -%} {%- assign is_external = false -%} {%- if safe_href == "" -%}{%- assign safe_href = "#" -%} {%- elsif low contains "https://" or low contains "http://" -%} {%- assign scheme = low | slice: 0, 8 -%} {%- assign scheme7 = low | slice: 0, 7 -%} {%- if scheme == "https://" or scheme7 == "http://" -%}{%- assign is_external = true -%}{%- else -%}{%- assign safe_href = "#" -%}{%- endif -%} {%- elsif low contains ":" -%} {%- assign scheme = low | slice: 0, 7 -%} {%- unless scheme == "mailto:" -%}{%- assign safe_href = "#" -%}{%- endunless -%} {%- endif -%} {%- assign first = safe_href | slice: 0 -%} {%- unless is_external or first == "#" or low contains "mailto:" or safe_href == "#" -%} {%- if first == "/" -%}{%- assign safe_href = safe_href | slice: 1, 2000 -%}{%- endif -%} {%- assign safe_href = page.root | append: safe_href -%} {%- endunless -%}
